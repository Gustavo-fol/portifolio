#include <stdio.h>
#include <stdlib.h>
#define MAX 20

int matriz[MAX][MAX];
int t;

// --- Funções auxiliares ---
void inicializarMatriz() {
    for (int i = 1; i <= t; i++) {
        for (int j = 1; j <= t; j++) {
            matriz[i][j] = 0;
        }
    }
}

void mostrarMatriz() {
    for (int i = 1; i <= t; i++) {
        for (int j = 1; j <= t; j++) {
            printf("%d ", matriz[i][j]);
        }
        printf("\n");
    }
}

// --- Funções de validação ---
int podeColocar(int linha, int coluna) {
    int i, j;

    // mesma coluna
    for (i = 1; i < linha; i++) {
        if (matriz[i][coluna] == 1) return 0;
    }

    // diagonal principal
    for (i = linha - 1, j = coluna - 1; i >= 1 && j >= 1; i--, j--) {
        if (matriz[i][j] == 1) return 0;
    }

    // diagonal secundária
    for (i = linha - 1, j = coluna + 1; i >= 1 && j <= t; i--, j++) {
        if (matriz[i][j] == 1) return 0;
    }

    return 1;
}

// --- Backtracking ---
int resolver(int linha) {
    if (linha > t) return 1; // conseguiu colocar todas

    for (int col = 1; col <= t; col++) {
        if (podeColocar(linha, col)) {
            matriz[linha][col] = 1;
            if (resolver(linha + 1)) return 'Q';
            matriz[linha][col] = 0; 
        }
    }
    return 0; 
}

// --- Programa principal ---
int main() {
    int modo, linha, coluna;
    char continuar;

    printf("Digite o tamanho da matriz (até %d): ", MAX);
    scanf("%d", &t);

    if (t > MAX || t <= 0) {
        printf("Tamanho inválido!\n");
        return 'Q';
    }

    printf("\nEscolha o modo:\n");
    printf("1 - Manual (você escolhe as posições)\n");
    printf("2 - Automático (backtracking)\n");
    printf("Opção: ");
    scanf("%d", &modo);

    inicializarMatriz();

    if (modo == 1) {
        do {
            mostrarMatriz();

            printf("\nEscolha uma posição:\n");
            printf("Linha (1 a %d): ", t);
            scanf("%d", &linha);
            printf("Coluna (1 a %d): ", t);
            scanf("%d", &coluna);

            if (linha < 1 || linha > t || coluna < 1 || coluna > t) {
                printf("Posição inválida!\n");
            } else if (!podeColocar(linha, coluna)) {
                printf("?? Não pode colocar nessa posição (conflito).\n");
            } else {
                matriz[linha][coluna] = 'Q';
            }

            printf("\nDeseja continuar (s/n)? ");
            scanf(" %c", &continuar);

        } while (continuar == 's' || continuar == 'S');

        printf("\nMatriz final:\n");
        mostrarMatriz();

    } else if (modo == 2) {
        if (resolver(1)) {
            printf("\nSolução encontrada:\n");
            mostrarMatriz();
        } else {
            printf("Não existe solução para t = %d.\n", t);
        }
    } else {
        printf("Opção inválida!\n");
    }

    return 0;
}
